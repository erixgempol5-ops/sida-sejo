Koneksi Database Supabase / PostgreSQL Online

3 Langkah Wajib Agar Supabase Terhubung Tanpa Error:
  1. Buka menu SQL Editor di dashboard Supabase, lalu tempel & klik Run pada skema SQL di bawah ini (wajib menyertakan GRANT ALL).
  2. Buka Project Settings → API di Supabase, salin Project URL (contoh: https://abcdefg.supabase.co).
  3. Salin Project API Keys → anon public (atau sb_publishable_...), lalu tempel di form bawah.
Skema Tabel & Hak Akses SQL (Jalankan di SQL Editor Supabase): 
-- 1. Buat tabel jika belum ada
CREATE TABLE IF NOT EXISTS public.sida_sejo_users (
  username TEXT PRIMARY KEY,
  password TEXT NOT NULL,
  role TEXT NOT NULL,
  name TEXT NOT NULL,
  rw TEXT,
  rt TEXT
);

CREATE TABLE IF NOT EXISTS public.sida_sejo_warga (
  id TEXT PRIMARY KEY,
  nik TEXT UNIQUE NOT NULL,
  kk TEXT NOT NULL,
  nama TEXT NOT NULL,
  tgl_lahir TEXT DEFAULT '',
  alamat TEXT NOT NULL,
  gender TEXT NOT NULL,
  pendidikan TEXT DEFAULT '-',
  status TEXT NOT NULL,
  no_hp TEXT,
  rw TEXT NOT NULL,
  rt TEXT NOT NULL,
  updated_at BIGINT DEFAULT 0
);

-- Tambahkan kolom pendidikan & tgl_lahir jika tabel sida_sejo_warga sudah dibuat sebelumnya
ALTER TABLE public.sida_sejo_warga ADD COLUMN IF NOT EXISTS pendidikan TEXT DEFAULT '-';
ALTER TABLE public.sida_sejo_warga ADD COLUMN IF NOT EXISTS tgl_lahir TEXT DEFAULT '';

-- 2. Berikan hak akses penuh ke peran anon & authenticated (atasi Error 42501)
GRANT USAGE ON SCHEMA public TO postgres, anon, authenticated, service_role;
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO postgres, anon, authenticated, service_role;
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public TO postgres, anon, authenticated, service_role;
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT ALL ON TABLES TO postgres, anon, authenticated, service_role;

-- 3. Nonaktifkan RLS agar akses langsung via API Key berjalan lancar
ALTER TABLE public.sida_sejo_users DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.sida_sejo_warga DISABLE ROW LEVEL SECURITY;

-- 4. Segarkan cache schema PostgREST
NOTIFY pgrst, 'reload schema';
